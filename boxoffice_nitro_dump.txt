707: export interface BoxOfficeNitroModule extends HybridObject<{ android: 'kotlin' }> {
708:   // Lifecycle
709:   configure(config: BoxOfficeConfig): Promise<CommandResult>
710:   start(): Promise<CommandResult>
711:   stop(): Promise<CommandResult>
712:   getStatus(): Promise<EngineStatus>
713: 
714:   // Search
715:   search(query: string, page: number, perPage: number, subjectType: SubjectType, version: ApiVersion): Promise<SearchResults>
716:   searchSuggestions(query: string, version: ApiVersion): Promise<SearchSuggestions>
717: 
718:   // Discovery
719:   getTrending(page: number, perPage: number, version: ApiVersion): Promise<TrendingResults>
720:   getHomepage(version: ApiVersion): Promise<HomepageContent>
721:   getHotContent(version: ApiVersion): Promise<HotContent>
722:   getPopularSearches(version: ApiVersion): Promise<PopularSearches>
723: 
724:   // Details
725:   getMovieDetails(urlOrItem: string, version: ApiVersion): Promise<MovieDetails>
726:   getTVSeriesDetails(urlOrItem: string, version: ApiVersion): Promise<TVSeriesDetails>
727:   getItemDetails(urlOrItem: string): Promise<V2ItemDetails>
728: 
729:   // Downloadable files
730:   getDownloadableFiles(item: SearchResultItem, subjectType: SubjectType, version: ApiVersion): Promise<DownloadableFiles>
731: 
732:   // Downloads
733:   downloadMovie(title: string, quality: string, captionLanguage: string, downloadDir: string, year: number): Promise<DownloadMovieResult>
734:   downloadTVSeries(title: string, season: number, episode: number, limit: number, quality: string, captionLanguage: string, downloadDir: string, autoMode: boolean): Promise<DownloadTVSeriesResult>
735:   getDownloadStatus(downloadId?: string): Promise<DownloadStatusList>
736:   cancelDownload(downloadId: string): Promise<CommandResult>
737: 
738:   // Recommendations
739:   getRecommendations(urlOrItem: string, page: number, perPage: number, version: ApiVersion): Promise<Recommendations>
740: 
741:   // Events
742:   addListener(event: 'onBoxOfficeStatusChange', callback: (event: StatusChangeEvent) => void): void
743:   addListener(event: 'onBoxOfficeCommandExecuted', callback: (event: CommandExecutedEvent) => void): void
744:   addListener(event: 'onBoxOfficeDownloadProgress', callback: (event: DownloadProgressEvent) => void): void
745:   addListener(event: 'onBoxOfficeError', callback: (event: ErrorEvent) => void): void
746:   removeListener(event: string, callback: () => void): void
747: }
